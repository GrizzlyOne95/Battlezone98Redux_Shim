    add_compile_options(/W4 /WX /permissive-)
endif()

target_link_libraries(frame_bridge_test_source PRIVATE d3d11 dxgi)

option(FRAME_BRIDGE_GPU_TESTS "Register hardware-dependent bridge producer tests" OFF)

    add_test(NAME frame_bridge_producer_gpu COMMAND frame_bridge_producer_gpu_tests)
    add_test(NAME frame_bridge_depth_gpu COMMAND frame_bridge_depth_gpu_tests)
    add_test(NAME frame_bridge_scene_gpu COMMAND frame_bridge_scene_gpu_tests)

if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/visual_enhancements_shader.h"
        COMMAND "${CMAKE_COMMAND}" "-DINPUT=${CMAKE_CURRENT_SOURCE_DIR}/visual_enhancements.hlsl"
            "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/visual_enhancements_shader.h" "-DSYMBOL=visualEnhancementsShader"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/Embed-Shader.cmake"
        DEPENDS visual_enhancements.hlsl Embed-Shader.cmake VERBATIM)
    add_library(frame_bridge_visuals STATIC visual_enhancements.cpp "${CMAKE_CURRENT_BINARY_DIR}/visual_enhancements_shader.h")

    target_link_libraries(frame_bridge_visuals PUBLIC d3d12 d3dcompiler)

    option(FRAME_BRIDGE_DXR "Build experimental inline-DXR visible geometry AO" OFF)
    add_library(frame_bridge_visible_rt STATIC visible_rt.cpp)
    target_link_libraries(frame_bridge_visible_rt PUBLIC d3d12 d3dcompiler)
    if(FRAME_BRIDGE_DXR)
        find_program(FRAME_BRIDGE_DXC dxc REQUIRED)
        foreach(entry Mesh Rays)
            string(TOLOWER "${entry}" lower)
            add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/visible_rt_${lower}.h"
                COMMAND "${FRAME_BRIDGE_DXC}" -T cs_6_5 -E "${entry}" -O3 -WX
                    -Fo "${CMAKE_CURRENT_BINARY_DIR}/visible_rt_${lower}.dxil" "${CMAKE_CURRENT_SOURCE_DIR}/visible_rt.hlsl"
                COMMAND "${CMAKE_COMMAND}" "-DINPUT=${CMAKE_CURRENT_BINARY_DIR}/visible_rt_${lower}.dxil"
                    "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/visible_rt_${lower}.h" "-DSYMBOL=visibleRt${entry}"
                    -P "${CMAKE_CURRENT_SOURCE_DIR}/Embed-Shader.cmake"
                DEPENDS visible_rt.hlsl Embed-Shader.cmake VERBATIM)
            target_sources(frame_bridge_visible_rt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/visible_rt_${lower}.h")
        endforeach()
        target_include_directories(frame_bridge_visible_rt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_compile_definitions(frame_bridge_visible_rt PRIVATE FRAME_BRIDGE_DXR=1)
        add_executable(frame_bridge_visible_rt_gpu_tests ../../tests/frame_bridge_visible_rt_gpu_tests.cpp)
        target_include_directories(frame_bridge_visible_rt_gpu_tests PRIVATE .)
        target_link_libraries(frame_bridge_visible_rt_gpu_tests PRIVATE frame_bridge_visible_rt)
        if(FRAME_BRIDGE_GPU_TESTS)
            add_test(NAME frame_bridge_visible_rt_gpu COMMAND frame_bridge_visible_rt_gpu_tests)
        endif()
    endif()
    option(FRAME_BRIDGE_RECORD_WINDOW "Record only the helper's own window with Windows Graphics Capture" OFF)
    add_executable(frame_bridge_presenter presenter.cpp gpu_motion.cpp gpu_blit.cpp streamline_fg.cpp window_recorder.cpp)
    target_link_libraries(frame_bridge_presenter PRIVATE frame_bridge_visuals frame_bridge_visible_rt d3d12 dxgi user32 d3dcompiler)

        find_path(FRAME_BRIDGE_CPPWINRT winrt/base.h PATHS

        target_link_libraries(frame_bridge_presenter PRIVATE windowsapp d3d11)
    endif()
    add_executable(frame_bridge_motion_gpu_tests ../../tests/frame_bridge_motion_gpu_tests.cpp gpu_motion.cpp)
